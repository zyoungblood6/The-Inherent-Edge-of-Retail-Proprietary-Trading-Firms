import pandas as pd
import numpy as np
import os

# =========================
# SETTINGS
# =========================

CSV_FILE = "NQ_data.csv"

OUTPUT_FILE = "mnq_10000_random_results.csv"
TRADE_LOG_FILE = "simulation_1_trades.csv"

# NEW: stores all 10,000 random LONG/SHORT paths
PATH_FILE = "mnq_10000_random_paths.npy"

# NEW: stores the day-level data needed to reconstruct trades
DAY_FILE = "mnq_random_clean_days.csv"

CONTRACTS = 4
TICK_SIZE = 0.25
TICKS = 200

POINT_DISTANCE = TICK_SIZE * TICKS
DOLLARS_PER_POINT_PER_CONTRACT = 2.0

TP_SL_DOLLARS = (
    POINT_DISTANCE
    * DOLLARS_PER_POINT_PER_CONTRACT
    * CONTRACTS
)

NUM_SIMULATIONS = 10000
BASE_SEED = 42

REQUIRED_START_TIME = "09:30:00"
REQUIRED_END_TIME = "16:00:00"


# =========================
# LOAD DATA
# =========================

if not os.path.exists(CSV_FILE):
    print(f"ERROR: Could not find {CSV_FILE}")
    exit()

df = pd.read_csv(
    CSV_FILE,
    sep=";",
    header=None,
    names=[
        "date",
        "time",
        "O",
        "H",
        "L",
        "C",
        "volume"
    ]
)

for col in ["O", "H", "L", "C", "volume"]:
    df[col] = pd.to_numeric(df[col], errors="coerce")

df["datetime"] = pd.to_datetime(
    df["date"].astype(str) + " " + df["time"].astype(str),
    errors="coerce"
)

df = df.dropna(
    subset=["datetime", "O", "H", "L", "C"]
)

df = df.sort_values("datetime").reset_index(drop=True)


# =========================
# BUILD ELIGIBLE DAYS
# =========================

eligible_days = []

for date, day in df.groupby(df["datetime"].dt.date):

    day = day.sort_values("datetime").reset_index(drop=True)

    first_time = day.iloc[0]["datetime"].strftime("%H:%M:%S")
    last_time = day.iloc[-1]["datetime"].strftime("%H:%M:%S")

    # Only full normal sessions
    if (
        first_time == REQUIRED_START_TIME
        and last_time == REQUIRED_END_TIME
    ):
        eligible_days.append(day)


print("==============================")
print("BUILDING ELIGIBLE TRADING DAYS")
print("==============================")

print(
    f"Total dates in dataset: "
    f"{df['date'].nunique()}"
)

print(
    f"Eligible full-session days: "
    f"{len(eligible_days)}"
)

print(
    f"Excluded holiday/early-close days: "
    f"{df['date'].nunique() - len(eligible_days)}"
)


# =========================
# PRECOMPUTE DAY RESULTS
# =========================

day_results = []

ambiguous_days = []

for day in eligible_days:

    date = day.iloc[0]["date"]

    entry_bar = day.iloc[0]

    entry = entry_bar["O"]
    entry_time = entry_bar["datetime"]

    long_target = entry + POINT_DISTANCE
    long_stop = entry - POINT_DISTANCE

    short_target = entry - POINT_DISTANCE
    short_stop = entry + POINT_DISTANCE

    post_entry = day[
        day["datetime"] > entry_time
    ]

    # =========================
    # LONG
    # =========================

    long_result = None
    long_exit_price = None
    long_exit_time = None

    for _, bar in post_entry.iterrows():

        hit_target = bar["H"] >= long_target
        hit_stop = bar["L"] <= long_stop

        if hit_target and hit_stop:

            long_result = "AMBIGUOUS"
            break

        elif hit_target:

            long_result = "WIN"
            long_exit_price = long_target
            long_exit_time = bar["datetime"]
            break

        elif hit_stop:

            long_result = "LOSS"
            long_exit_price = long_stop
            long_exit_time = bar["datetime"]
            break

    if long_result is None:

        final_close = day.iloc[-1]["C"]

        long_result = "TIME_EXIT"
        long_exit_price = final_close
        long_exit_time = day.iloc[-1]["datetime"]

        long_pnl = (
            (final_close - entry)
            * DOLLARS_PER_POINT_PER_CONTRACT
            * CONTRACTS
        )

    elif long_result == "WIN":

        long_pnl = TP_SL_DOLLARS

    elif long_result == "LOSS":

        long_pnl = -TP_SL_DOLLARS

    else:

        long_pnl = np.nan


    # =========================
    # SHORT
    # =========================

    short_result = None
    short_exit_price = None
    short_exit_time = None

    for _, bar in post_entry.iterrows():

        hit_target = bar["L"] <= short_target
        hit_stop = bar["H"] >= short_stop

        if hit_target and hit_stop:

            short_result = "AMBIGUOUS"
            break

        elif hit_target:

            short_result = "WIN"
            short_exit_price = short_target
            short_exit_time = bar["datetime"]
            break

        elif hit_stop:

            short_result = "LOSS"
            short_exit_price = short_stop
            short_exit_time = bar["datetime"]
            break

    if short_result is None:

        final_close = day.iloc[-1]["C"]

        short_result = "TIME_EXIT"
        short_exit_price = final_close
        short_exit_time = day.iloc[-1]["datetime"]

        short_pnl = (
            (entry - final_close)
            * DOLLARS_PER_POINT_PER_CONTRACT
            * CONTRACTS
        )

    elif short_result == "WIN":

        short_pnl = TP_SL_DOLLARS

    elif short_result == "LOSS":

        short_pnl = -TP_SL_DOLLARS

    else:

        short_pnl = np.nan


    # =========================
    # SAVE DAY
    # =========================

    if (
        long_result == "AMBIGUOUS"
        or short_result == "AMBIGUOUS"
    ):
        ambiguous_days.append(date)

    day_results.append({

        "date": date,

        "entry_time": entry_time,
        "entry": entry,

        "long_pnl": long_pnl,
        "short_pnl": short_pnl,

        "long_result": long_result,
        "short_result": short_result,

        "long_exit_price": long_exit_price,
        "short_exit_price": short_exit_price,

        "long_exit_time": long_exit_time,
        "short_exit_time": short_exit_time

    })


day_results = pd.DataFrame(day_results)


# =========================
# REMOVE AMBIGUOUS DAYS
# =========================

clean_days = day_results[
    day_results["long_pnl"].notna()
    & day_results["short_pnl"].notna()
].reset_index(drop=True)


print(
    f"Ambiguous days: "
    f"{len(ambiguous_days)}"
)

print(
    f"Final eligible days used: "
    f"{len(clean_days)}"
)


# =========================
# SAVE CLEAN DAY DATA
# =========================

clean_days.to_csv(
    DAY_FILE,
    index=False
)


# =========================
# RUN 10,000 SIMULATIONS
# =========================

rng = np.random.default_rng(BASE_SEED)

long_pnl = clean_days["long_pnl"].to_numpy()
short_pnl = clean_days["short_pnl"].to_numpy()

simulation_results = []

simulation_1_trades = []

# NEW:
# Each row = one simulation
# Each column = one trading day
# 0 = SHORT
# 1 = LONG
all_directions = np.empty(
    (NUM_SIMULATIONS, len(clean_days)),
    dtype=np.int8
)


print("\n==============================")
print("RUNNING RANDOM SIMULATIONS")
print("==============================")


for simulation in range(1, NUM_SIMULATIONS + 1):

    # 0 = SHORT
    # 1 = LONG

    directions = rng.integers(
        0,
        2,
        size=len(clean_days)
    )

    # Store exact random path
    all_directions[simulation - 1] = directions

    pnl = np.where(
        directions == 1,
        long_pnl,
        short_pnl
    )

    wins = np.sum(
        pnl == TP_SL_DOLLARS
    )

    losses = np.sum(
        pnl == -TP_SL_DOLLARS
    )

    time_exits = (
        len(pnl)
        - wins
        - losses
    )

    total_pnl = pnl.sum()

    average_pnl = pnl.mean()


    # =========================
    # SAVE FULL TRADE LOG
    # FOR SIMULATION #1
    # =========================

    if simulation == 1:

        for i in range(len(clean_days)):

            day = clean_days.iloc[i]

            if directions[i] == 1:

                direction = "LONG"

                target = day["entry"] + POINT_DISTANCE
                stop = day["entry"] - POINT_DISTANCE

                result = day["long_result"]
                exit_price = day["long_exit_price"]
                exit_time = day["long_exit_time"]

            else:

                direction = "SHORT"

                target = day["entry"] - POINT_DISTANCE
                stop = day["entry"] + POINT_DISTANCE

                result = day["short_result"]
                exit_price = day["short_exit_price"]
                exit_time = day["short_exit_time"]


            simulation_1_trades.append({

                "simulation": 1,

                "date": day["date"],

                "direction": direction,

                "entry_time": day["entry_time"],

                "entry": day["entry"],

                "target": target,

                "stop": stop,

                "result": result,

                "exit_time": exit_time,

                "exit_price": exit_price,

                "pnl": pnl[i]

            })


    simulation_results.append({

        "simulation": simulation,

        "trades": len(pnl),

        "wins": wins,

        "losses": losses,

        "time_exits": time_exits,

        "win_rate": (
            wins
            / (wins + losses)
            * 100
            if (wins + losses) > 0
            else 0
        ),

        "total_pnl": total_pnl,

        "average_pnl": average_pnl

    })


# =========================
# SAVE ALL RANDOM PATHS
# =========================

np.save(
    PATH_FILE,
    all_directions
)


# =========================
# SAVE 10,000 RESULTS
# =========================

results = pd.DataFrame(
    simulation_results
)

results.to_csv(
    OUTPUT_FILE,
    index=False
)


# =========================
# SAVE SIMULATION #1
# =========================

trade_log = pd.DataFrame(
    simulation_1_trades
)

trade_log.to_csv(
    TRADE_LOG_FILE,
    index=False
)


# =========================
# FINAL RESULTS
# =========================

print("\n==============================")
print("10,000 RANDOM NULL STRATEGIES")
print("==============================")

print(
    f"Simulations: "
    f"{len(results):,}"
)

print(
    f"Trades per simulation: "
    f"{len(clean_days):,}"
)

print(
    f"Mean total P&L: "
    f"${results.total_pnl.mean():,.2f}"
)

print(
    f"Median total P&L: "
    f"${results.total_pnl.median():,.2f}"
)

print(
    f"Std. dev. total P&L: "
    f"${results.total_pnl.std():,.2f}"
)

print(
    f"5th percentile: "
    f"${results.total_pnl.quantile(.05):,.2f}"
)

print(
    f"95th percentile: "
    f"${results.total_pnl.quantile(.95):,.2f}"
)

print(
    f"Mean average trade: "
    f"${results.average_pnl.mean():,.4f}"
)

print(
    f"Median average trade: "
    f"${results.average_pnl.median():,.4f}"
)

print(
    f"Mean TP/SL win rate: "
    f"{results.win_rate.mean():.2f}%"
)

print(
    f"Profitable simulations: "
    f"{(results.total_pnl > 0).sum():,}"
    f" ({(results.total_pnl > 0).mean() * 100:.2f}%)"
)

print(
    f"Losing simulations: "
    f"{(results.total_pnl < 0).sum():,}"
    f" ({(results.total_pnl < 0).mean() * 100:.2f}%)"
)

print("\nSaved:")
print(OUTPUT_FILE)
print(TRADE_LOG_FILE)
print(PATH_FILE)
print(DAY_FILE)
